public class SemanticTest {
    public static void main(String[] a) {
        System.out.println(new TestClass().run());
    }
}

class TestClass {
    int x;
    boolean b;

    public int run() {
        x = b;          // assigning boolean to int
        x = unknown;    // undeclared variable
        return b;       // return boolean instead of int
    }

    public int add(int a, int b) {
        return a + b;
    }

    public int returnAdd() {
        return this.add(true, 1); // wrong argument type, boolean instead of int
    }
}
